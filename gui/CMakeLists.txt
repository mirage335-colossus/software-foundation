    math(EXPR license_last "${license_count} - 1")

        string(JSON license_file GET "${lock}" redistribution license_files ${index})
        string(JSON license_hash ERROR_VARIABLE missing_license GET "${lock}" files "${license_file}")

            message(FATAL_ERROR "Redistribution license must be in the verified input inventory")
        endif()
    endforeach()

foreach(index RANGE 0 ${last_file})
    string(JSON relative MEMBER "${lock}" files ${index})
    string(JSON expected GET "${lock}" files "${relative}")
    set(path "${FOUNDATION_GUI_SOURCE}/${relative}")
    if(NOT EXISTS "${path}")
        message(FATAL_ERROR "Pinned GUI input missing: ${relative}")
    endif()
    file(SHA256 "${path}" actual)
    if(NOT actual STREQUAL expected)
        message(FATAL_ERROR "Pinned GUI input differs: ${relative}. Review upgrade and lock together.")
    endif()
    list(APPEND verified_files "${path}")
endforeach()

function(foundation_gui_patch source patch output)
    get_filename_component(output_directory "${CMAKE_CURRENT_BINARY_DIR}/${output}" DIRECTORY)
    file(MAKE_DIRECTORY "${output_directory}")
    execute_process(COMMAND "${Python3_EXECUTABLE}" -B "${CMAKE_CURRENT_SOURCE_DIR}/patches/apply.py"
        "${FOUNDATION_GUI_SOURCE}/${source}" "${CMAKE_CURRENT_SOURCE_DIR}/patches/${patch}"
        "${CMAKE_CURRENT_BINARY_DIR}/${output}" COMMAND_ERROR_IS_FATAL ANY)
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
        "${CMAKE_CURRENT_SOURCE_DIR}/patches/${patch}" "${CMAKE_CURRENT_SOURCE_DIR}/patches/apply.py")
endfunction()
# Continue a verified patch sequence without exposing intermediate assets in the
# installed web directory. The exact patcher preserves unchanged output mtimes.
function(foundation_gui_repatch source patch output)
    get_filename_component(output_directory "${CMAKE_CURRENT_BINARY_DIR}/${output}" DIRECTORY)
    file(MAKE_DIRECTORY "${output_directory}")
    execute_process(COMMAND "${Python3_EXECUTABLE}" -B "${CMAKE_CURRENT_SOURCE_DIR}/patches/apply.py"
        "${CMAKE_CURRENT_BINARY_DIR}/${source}" "${CMAKE_CURRENT_SOURCE_DIR}/patches/${patch}"
        "${CMAKE_CURRENT_BINARY_DIR}/${output}" COMMAND_ERROR_IS_FATAL ANY)
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
        "${CMAKE_CURRENT_SOURCE_DIR}/patches/${patch}")
endfunction()
function(foundation_gui_check name target labels)
    set_target_properties(${target} PROPERTIES EXCLUDE_FROM_ALL TRUE)
    add_test(NAME "foundation.gui.${name}" COMMAND ${target})
    set_tests_properties("foundation.gui.${name}" PROPERTIES LABELS "gui;${labels}" TIMEOUT 90)
    foundation_test_prerequisites("foundation.gui.${name}" ${target})
endfunction()
function(foundation_gui_sdl_display_test name)
    # Qualify the declared native service; minimal SDKs need no dummy driver.
    if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
        set(driver x11)
    elseif(CMAKE_SYSTEM_NAME STREQUAL "Windows")
        set(driver windows)
    else()
        message(FATAL_ERROR "Declare the native SDL video driver for this qualification host")
    endif()
    set_tests_properties(${name} PROPERTIES ENVIRONMENT "SDL_VIDEODRIVER=${driver}"
        RESOURCE_LOCK foundation_native_display)
endfunction()
