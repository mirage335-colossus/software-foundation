# Enable from the parent with FOUNDATION_BUILD_GUI and FOUNDATION_GUI_SOURCE.
# No implicit fetching and no toolkit flags escape the adapter targets.
set(FOUNDATION_GUI_SOURCE "" CACHE PATH "Verified gui-boundary source checkout or archive")
option(FOUNDATION_GUI_FLTK "Build the optional FLTK native host" OFF)
option(FOUNDATION_GUI_HOST_TESTS "Run tests requiring native host resources" OFF)
if(NOT IS_DIRECTORY "${FOUNDATION_GUI_SOURCE}/include/gui")
    message(FATAL_ERROR "Set FOUNDATION_GUI_SOURCE to the pinned gui-boundary source directory; see docs/gui-boundary.md")
endif()
get_filename_component(FOUNDATION_GUI_SOURCE "${FOUNDATION_GUI_SOURCE}" REALPATH)
set(lock_path "${CMAKE_CURRENT_SOURCE_DIR}/../third_party/gui-boundary.lock.json")
file(READ "${lock_path}" lock)
string(JSON file_count LENGTH "${lock}" files)
math(EXPR last_file "${file_count} - 1")
set(verified_files)
foreach(index RANGE 0 ${last_file})
    string(JSON relative MEMBER "${lock}" files ${index})
    string(JSON expected GET "${lock}" files "${relative}")
    set(path "${FOUNDATION_GUI_SOURCE}/${relative}")
    if(NOT EXISTS "${path}")
        message(FATAL_ERROR "Pinned GUI input missing: ${relative}")
    endif()
    file(SHA256 "${path}" actual)
    if(NOT actual STREQUAL expected)
        message(FATAL_ERROR "Pinned GUI input differs: ${relative}. Review upgrade and lock together.")
    endif()
    list(APPEND verified_files "${path}")
endforeach()
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${lock_path}" ${verified_files})
find_package(Threads REQUIRED)
add_library(foundation_gui_boundary INTERFACE)
target_include_directories(foundation_gui_boundary SYSTEM INTERFACE "${FOUNDATION_GUI_SOURCE}/include")
target_compile_features(foundation_gui_boundary INTERFACE cxx_std_20)
target_link_libraries(foundation_gui_boundary INTERFACE Threads::Threads)

# Upstream hosts bind their example directly. Apply one fail-closed composition
# replacement to a verified file in the build directory. Platform code stays
# wholly upstream. This is an integration patch, not a second maintained host.
function(foundation_bind_gui_host output source original_include)
    file(READ "${source}" host)
    string(FIND "${host}" "${original_include}" location)
    if(location EQUAL -1)
        message(FATAL_ERROR "GUI host composition include changed; review integration patch")
    endif()
    string(REPLACE "${original_include}" "" without_include "${host}")
    string(LENGTH "${host}" before)
    string(LENGTH "${without_include}" after)
    string(LENGTH "${original_include}" include_length)
    math(EXPR removed "${before} - ${after}")
    if(NOT removed EQUAL include_length)
        message(FATAL_ERROR "Expected exactly one GUI application composition include")
    endif()
    string(REPLACE "${original_include}"
        "#include \"shared/application.hpp\"\nusing Example = foundation::ui::Application;"
        host "${host}")
    file(WRITE "${output}" "${host}")
endfunction()
